# Qt 5 QML API shims for unported Sailfish apps on Keel (README.md). QML only,
# no plugin. The modules go into their own import directory, which keel-shell
# puts first on QML_IMPORT_PATH for Keel apps; they are NOT installed into
# Qt 6's QML directory, so other Qt 6 programs never see them.
#
# Also the Qt 5 source rewrite (src/qt5source.h, README.md "Qt 5 JavaScript in
# app sources"): a static library that keel/sailfishapp links into
# libkeel-sailfishapp and the corpus harness compiles in. It is built
# whatever KEEL_BUILD_QT5COMPAT says, since SailfishApp needs it.
add_library(keel-qt5compat-source STATIC src/qt5source.cpp src/qt5source.h)
set_target_properties(keel-qt5compat-source PROPERTIES
    POSITION_INDEPENDENT_CODE ON AUTOMOC OFF
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(keel-qt5compat-source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(keel-qt5compat-source PUBLIC Qt6::Core Qt6::Qml)
if(BUILD_TESTING)
    qt_add_executable(tst_keel_qt5compat_source tests/source/tst_qt5source.cpp)
    target_link_libraries(tst_keel_qt5compat_source PRIVATE keel-qt5compat-source Qt6::Test Qt6::Qml)
    add_test(NAME keel_qt5compat_source COMMAND tst_keel_qt5compat_source)
    set_tests_properties(keel_qt5compat_source PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()

option(KEEL_BUILD_QT5COMPAT "Build the Qt 5 QML API shims (keel/qt5compat)" ON)
if(NOT KEEL_BUILD_QT5COMPAT)
    return()
endif()

# Staged in the build tree (tests and the corpus harness import from here).
set(KEEL_QT5COMPAT_OUTPUT_DIR "${CMAKE_BINARY_DIR}/qml-qt5compat")
set(KEEL_QT5COMPAT_OUTPUT_DIR "${KEEL_QT5COMPAT_OUTPUT_DIR}" PARENT_SCOPE)
set(KEEL_QT5COMPAT_INSTALL_DIR "${CMAKE_INSTALL_LIBDIR}/keel/qt5compat"
    CACHE PATH "Import directory for the Qt 5 QML API shims (keel-shell adds it)")

# Static modules: QtGraphicalEffects, QtFeedback, QtMultimedia.5 and the
# shared KeelQt5Compat.QtQuick types.
file(GLOB_RECURSE _qt5compat_files RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/qml"
     CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/qml/*")
foreach(_f IN LISTS _qt5compat_files)
    configure_file("qml/${_f}" "${KEEL_QT5COMPAT_OUTPUT_DIR}/${_f}" COPYONLY)
endforeach()

# QtQuick.2.0 .. QtQuick.2.14: Qt looks for version-qualified module
# directories before the plain one, so `import QtQuick 2.<minor>` loads these.
# Each re-exports the real Qt Quick at 2.15 (no QtQuick.2.15 or QtQuick.2
# directory exists, so that import reaches Qt's own module) plus the Qt 5
# types. Only `import QtQuick 2.15` and `6.x` bypass the shims. Keel's Silica
# ports import QtQuick 2.0 to 2.6 (Silica's BSD QML), so under keel-shell they
# go through these directories too; tests/core/tst_qtquick_identity.qml checks
# that they still get Qt's own Qt Quick types, and
# keel_qt5compat_silica_regression runs Keel's Silica tests this way.
# There is deliberately no `module QtQuick` line: with it, Qt 6.8 refuses to
# register Qt Quick's own types afterwards ("Namespace 'QtQuick' has already
# been used for type registration"); Qt 6.4 accepts both forms.
# Keel.SilicaScreen (keel/silica: Silica's Screen singleton, a superset of
# Qt Quick's attached Screen) comes after Qt Quick: of a qmldir's imports the
# last one with a name wins, so `Screen` under `import QtQuick 2.x` is
# Silica's, as in Qt 5 (where Qt Quick 2.x had no Screen) even when the file
# imports QtQuick before Sailfish.Silica (tests/silicascreen).
foreach(_minor RANGE 0 14)
    file(CONFIGURE OUTPUT "${KEEL_QT5COMPAT_OUTPUT_DIR}/QtQuick.2.${_minor}/qmldir" CONTENT
"# Generated by keel/qt5compat/CMakeLists.txt. See keel/qt5compat/README.md.
import QtQuick 2.15
import KeelQt5Compat.QtQuick 1.0
import Keel.SilicaScreen 1.0
")
endforeach()

install(DIRECTORY "${KEEL_QT5COMPAT_OUTPUT_DIR}/" DESTINATION "${KEEL_QT5COMPAT_INSTALL_DIR}")

if(NOT BUILD_TESTING)
    return()
endif()

# Tests: qmltestrunner (offscreen) with the shims first on the import path.
# KEEL_QT5COMPAT_TEST_ENV adds environment entries (e.g. QML_IMPORT_PATH for
# host extras, LD_LIBRARY_PATH, QT_PLUGIN_PATH, GST_PLUGIN_PATH).
set(KEEL_QT5COMPAT_TEST_ENV "" CACHE STRING "Extra environment (;-list) for the qt5compat tests")
set(KEEL_QT5COMPAT_TEST_MULTIMEDIA OFF CACHE BOOL
    "Run the QtMultimedia 5 shim test (needs Qt 6 Multimedia with a working backend)")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/runtime" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
set(_qt5compat_env
    "QT_QPA_PLATFORM=offscreen"
    "XDG_RUNTIME_DIR=${CMAKE_CURRENT_BINARY_DIR}/runtime"
    ${KEEL_QT5COMPAT_TEST_ENV})

# Keel's QML directory for Keel.SilicaScreen, which the shims import.
function(keel_qt5compat_test name dir)
    add_test(NAME ${name}
        COMMAND ${KEEL_QMLTESTRUNNER} -import "${KEEL_QT5COMPAT_OUTPUT_DIR}"
                -import "${KEEL_QML_OUTPUT_DIR}"
                -input "${CMAKE_CURRENT_SOURCE_DIR}/tests/${dir}")
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "${_qt5compat_env}" TIMEOUT 120)
endfunction()

keel_qt5compat_test(keel_qt5compat_core core)

# QtGraphicalEffects needs Qt5Compat.GraphicalEffects in Qt's QML directory.
get_target_property(_keel_qmake Qt6::qmake IMPORTED_LOCATION)
execute_process(COMMAND "${_keel_qmake}" -query QT_INSTALL_QML
                OUTPUT_VARIABLE _qt5compat_qt_qml OUTPUT_STRIP_TRAILING_WHITESPACE)
if(EXISTS "${_qt5compat_qt_qml}/Qt5Compat/GraphicalEffects/qmldir")
    keel_qt5compat_test(keel_qt5compat_graphicaleffects graphicaleffects)
else()
    message(STATUS "keel/qt5compat: Qt5Compat.GraphicalEffects not installed; "
                   "skipping keel_qt5compat_graphicaleffects")
endif()

if(KEEL_QT5COMPAT_TEST_MULTIMEDIA)
    keel_qt5compat_test(keel_qt5compat_multimedia multimedia)
endif()

# The Qt 5 Camera shim on a machine without a camera device. Needs Qt 6
# Multimedia's QML module and a loadable backend (Qt 6.4 aborts when it lists
# cameras without one), but no camera, and plays nothing. Registered when Qt
# has the module, or on request when it comes from elsewhere (QML_IMPORT_PATH
# and QT_PLUGIN_PATH in KEEL_QT5COMPAT_TEST_ENV).
set(KEEL_QT5COMPAT_TEST_CAMERA OFF CACHE BOOL
    "Run the QtMultimedia 5 Camera shim test although Qt's QML directory lacks QtMultimedia")
if(KEEL_QT5COMPAT_TEST_CAMERA OR EXISTS "${_qt5compat_qt_qml}/QtMultimedia/qmldir")
    keel_qt5compat_test(keel_qt5compat_camera camera)
else()
    message(STATUS "keel/qt5compat: QtMultimedia QML module not installed; "
                   "skipping keel_qt5compat_camera")
endif()

# Keel's own Silica tests with the shims on the import path, as keel-shell
# runs apps: the QtQuick.2.<minor> re-export must not change Keel.
if(TARGET silicaplugin)
    # `Screen` is Silica's under the shims whatever the import order.
    add_test(NAME keel_qt5compat_silica_screen
        COMMAND ${KEEL_QMLTESTRUNNER} -import "${KEEL_QT5COMPAT_OUTPUT_DIR}"
                -import "${KEEL_QML_OUTPUT_DIR}"
                -input "${CMAKE_CURRENT_SOURCE_DIR}/tests/silicascreen")
    set_tests_properties(keel_qt5compat_silica_screen PROPERTIES
        ENVIRONMENT "${_qt5compat_env};QML_IMPORT_PATH=${KEEL_QT5COMPAT_OUTPUT_DIR}:${KEEL_QML_OUTPUT_DIR}"
        TIMEOUT 120)
    add_test(NAME keel_qt5compat_silica_regression
        COMMAND ${KEEL_QMLTESTRUNNER} -import "${KEEL_QT5COMPAT_OUTPUT_DIR}"
                -import "${KEEL_QML_OUTPUT_DIR}"
                -input "${CMAKE_CURRENT_SOURCE_DIR}/../silica/tests/qml")
    set_tests_properties(keel_qt5compat_silica_regression PROPERTIES
        ENVIRONMENT "${_qt5compat_env};QML_IMPORT_PATH=${KEEL_QT5COMPAT_OUTPUT_DIR}:${KEEL_QML_OUTPUT_DIR}"
        TIMEOUT 300)
endif()
